local guiLabel, lastGuiScan = nil, 0

local function parseHP(text)
    text = text:gsub("<[^>]+>", ""):gsub(",", "")
    local a, b = text:match("(%d+%.?%d*)%s*/%s*(%d+%.?%d*)")
    if a then return tonumber(a), tonumber(b) end
end

local function readGui()
    if guiLabel and guiLabel.Parent then
        local a, b = parseHP(guiLabel.Text)
        if a then return a, b end
    end
    guiLabel = nil
    if os.clock() - lastGuiScan < 2 then return end
    lastGuiScan = os.clock()

    local pg = lp:FindFirstChild("PlayerGui")
    if not pg then return end
    local best, bestScore
    for _, o in ipairs(pg:GetDescendants()) do
        if (o:IsA("TextLabel") or o:IsA("TextButton")) and o.Visible and o.Text ~= "" then
            local pos = o.AbsolutePosition
            if pos.X < 500 and pos.Y < 300 then
                local a, b = parseHP(o.Text)
                if a and b and b > 0 and a <= b then
                    local score = pos.X + pos.Y
                    local n = (o.Name .. (o.Parent and o.Parent.Name or "")):lower()
                    if n:find("health") or n:find("hp") then score -= 1000 end
                    if not bestScore or score < bestScore then best, bestScore = o, score end
                end
            end
        end
    end
    if best then
        guiLabel = best
        return parseHP(best.Text)
    end
end

-- baca MP dari GUI: cari label "angka/angka" bernama mp/mana/energy,
-- kalau tidak ada ambil label bar berikutnya setelah HP (di kiri atas)
local function readMpGui()
    if mpGuiLabel and mpGuiLabel.Parent then
        local a, b = parseHP(mpGuiLabel.Text)
        if a then return a, b end
    end
    mpGuiLabel = nil
    if os.clock() - lastMpScan < 2 then return end
    lastMpScan = os.clock()

    local pg = lp:FindFirstChild("PlayerGui")
    if not pg then return end
    local best, bestScore
    for _, o in ipairs(pg:GetDescendants()) do
        if o ~= guiLabel and (o:IsA("TextLabel") or o:IsA("TextButton")) and o.Visible and o.Text ~= "" then
            local pos = o.AbsolutePosition
            if pos.X < 500 and pos.Y < 300 then
                local a, b = parseHP(o.Text)
                if a and b and b > 0 and a <= b then
                    local n = (o.Name .. (o.Parent and o.Parent.Name or "")):lower()
                    if not (n:find("health") or n:find("hp")) then
                        local score = pos.X + pos.Y
                        if n:find("mp") or n:find("mana") or n:find("energy") then score -= 1000 end
                        if not bestScore or score < bestScore then best, bestScore = o, score end
                    end
                end
            end
        end
    end
    if best then
        mpGuiLabel = best
        return parseHP(best.Text)
    end
end

local HP_NAMES = { "Health", "HP", "CurrentHealth", "hp", "health", "Hp" }
local MAX_NAMES = { "MaxHealth", "MaxHP", "maxHealth", "MaxHp", "maxHp", "maxhp" }
local MP_NAMES = { "Mana", "MP", "CurrentMana", "Energy", "mana", "mp", "energy", "Mp" }
local MPMAX_NAMES = { "MaxMana", "MaxMP", "maxMana", "MaxEnergy", "maxMp", "maxmp", "maxEnergy", "MaxMp" }

local function readAttr(names, maxNames)
    local c = lp.Character
    local hum = c and c:FindFirstChildOfClass("Humanoid")
    local holders = { c, hum, lp, lp:FindFirstChild("leaderstats"), lp:FindFirstChild("Data") }
    for _, h in ipairs(holders) do
        if h then
            for _, n in ipairs(names) do
                local v = h:GetAttribute(n)
                if type(v) ~= "number" then
                    local ch = h:FindFirstChild(n)
                    if ch and (ch:IsA("NumberValue") or ch:IsA("IntValue")) then v = ch.Value end
                end
                if type(v) == "number" then
                    local mx
                    for _, mn in ipairs(maxNames) do
                        local m = h:GetAttribute(mn)
                        if type(m) ~= "number" then
                            local ch = h:FindFirstChild(mn)
                            if ch and (ch:IsA("NumberValue") or ch:IsA("IntValue")) then m = ch.Value end
                        end
                        if type(m) == "number" then mx = m break end
                    end
                    return v, mx
                end
            end
        end
    end
end

task.spawn(function()
    while true do
        if gui and not gui.Parent then break end
        local hp, mx = readHP()
        if hp then
            if lastHp and hp < lastHp then damaged = true end
            lastHp, curHp, curMax = hp, hp, mx
            if running and lowHp() then recovering = true end
            if hpLabel then
                local txt = string.format("HP %d%s [%s]", hp, mx and ("/" .. math.floor(mx)) or "", SRC_NAMES[hpSrc])
                if mx and mx <= num("hplow") then txt = txt .. " (MaxHP<=batas!)" end
                hpLabel.Text = txt
            end
        else
            curHp, curMax = nil, nil
            if hpLabel then hpLabel.Text = "HP tidak terbaca [" .. SRC_NAMES[hpSrc] .. "]" end
        end

        -- MP: berhenti pakai E di bawah "MP stop", lanjut lagi saat mencapai "MP lanjut"
        local mp, mmx = readMP()
        if mp then
            curMp, curMpMax = mp, mmx
            local low = num("mplow")
            local okv = math.max(num("mpok"), low + 1)
            if mmx and mmx < okv then okv = mmx end
            if mp < low then
                mpRest = true
            elseif mpRest and mp >= okv - 0.5 then
                mpRest = false
            end
            if mpLabel then
                mpLabel.Text = string.format("MP %d%s%s", mp, mmx and ("/" .. math.floor(mmx)) or "", mpRest and "  (pulih, tanpa E)" or "")
            end
        else
            curMp, curMpMax = nil, nil
            mpRest = false
            if mpLabel then mpLabel.Text = "MP tidak terbaca" end
        end
        task.wait(0.1)
    end
end)

local function mobPart(m)
    return m:FindFirstChild("HumanoidRootPart") or m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart", true)
end

local function nameMatches(m, list)
    local n = m.Name:lower()
    local h = m:FindFirstChildOfClass("Humanoid")
    local dn = h and h.DisplayName:lower() or ""
    for _, key in ipairs(list) do
        if n:find(key, 1, true) or dn:find(key, 1, true) then return true end
    end
    return false
end

-- true jika pos berada dalam radius scan dari titik 1 atau titik 2.
-- Kalau kedua titik belum diset -> semua area dianggap valid.
local function inArea(pos, extra)
    extra = extra or 0
    local any = false
    local r = num("scanr") + extra
    for i = 1, 2 do
        local p = points[i]
        if p then
            any = true
            if (pos - p).Magnitude <= r then return true end
        end
    end
    return not any
end

-- DINDING MERAH di tepi radius scan (visual saja: tidak bisa ditabrak,
-- tidak ikut raycast, hanya terlihat di layar kita)

local WALL_HEIGHT = 80
local WALL_COLOR = Color3.fromRGB(255, 30, 30)
local WALL_TRANSPARENCY = 0.88 -- makin dekat 1 makin transparan
local CENTER_SIZE = 5          -- diameter bola merah titik tengah (stud)

local function wallSignature()
    return string.format("%s|%s|%s|%s", tostring(num("scanr")), tostring(points[1]), tostring(points[2]), tostring(wallsOn))
end

local function clearWalls()
    if wallFolder then
        wallFolder:Destroy()
        wallFolder = nil
    end
end

local function drawWalls()
    clearWalls()
    wallSig = wallSignature()
    if not wallsOn then return end
    local cam = workspace.CurrentCamera
    local R = num("scanr")
    if not cam or R <= 0 or not (points[1] or points[2]) then return end

    wallFolder = Instance.new("Folder")
    wallFolder.Name = "PHWalls"
    wallFolder.Parent = cam

    local n = math.clamp(math.floor(R * 2 * math.pi / 12), 24, 96)
    local segLen = 2 * math.pi * R / n
    for i = 1, 2 do
        local c = points[i]
        if c then
            for k = 0, n - 1 do
                local a = k / n * 2 * math.pi
                local pos = c + Vector3.new(math.cos(a) * R, WALL_HEIGHT / 4, math.sin(a) * R)

                -- jika segmen ini berada di dalam lingkaran titik lainnya (area bertemu) -> jangan digambar
                local inside = false
                local other = points[3 - i]
                if other then
                    local dx, dz = pos.X - other.X, pos.Z - other.Z
                    inside = math.sqrt(dx * dx + dz * dz) < R
                end
                if not inside then
                    local p = Instance.new("Part")
                    p.Anchored = true
                    p.CanCollide = false
                    p.CanQuery = false
                    p.CanTouch = false
                    p.CastShadow = false
                    p.Material = Enum.Material.Neon
                    p.Color = WALL_COLOR
                    p.Transparency = WALL_TRANSPARENCY
                    p.Size = Vector3.new(segLen + 0.3, WALL_HEIGHT, 0.5)
                    p.CFrame = CFrame.lookAt(pos, Vector3.new(c.X, pos.Y, c.Z))
                    p.Parent = wallFolder
                end
            end

            -- bola merah penanda titik tengah
            local ball = Instance.new("Part")
            ball.Shape = Enum.PartType.Ball
            ball.Anchored = true
            ball.CanCollide = false
            ball.CanQuery = false
            ball.CanTouch = false
            ball.CastShadow = false
            ball.Material = Enum.Material.Neon
            ball.Color = WALL_COLOR
            ball.Transparency = 0.15
            ball.Size = Vector3.new(CENTER_SIZE, CENTER_SIZE, CENTER_SIZE)
            ball.Position = c
            ball.Parent = wallFolder
        end
    end
end

local function alive(e)
    local h = e.m and e.m.Parent and e.m:FindFirstChildOfClass("Humanoid")
    return e.m ~= nil and e.m.Parent ~= nil and (not h or h.Health > 0)
end

local cache = { t = -1e9, hounds = {}, avoid = {} }

local function invalidate()
    cache.t = -1e9
end

local function mark(m, color)
    if not hlFolder then return end
    local h = hlMap[m]
    if not h or not h.Parent then
        h = Instance.new("Highlight")
        h.Adornee = m
        h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        h.FillTransparency = 0.6
        h.OutlineTransparency = 0
        h.Parent = hlFolder
        hlMap[m] = h
    end
    h.FillColor = color
    h.OutlineColor = color
end

-- scan penuh workspace hanya tiap "Scan ulang" detik (default 10), kecuali force
local function scan(force)
    if not force and os.clock() - cache.t < math.max(num("scanint"), 1) then return cache end
    cache.t = os.clock()
    local hounds, avoid, seen = {}, {}, {}
    local keys = targetKeys()
    local avoidExtra = num("avoid") -- monster yang dihindari dideteksi sedikit lebih luas dari area

    local function add(list, m, extra)
        if seen[m] then return end
        local h = m:FindFirstChildOfClass("Humanoid")
        local p = mobPart(m)
        if p and (not h or h.Health > 0) and inArea(p.Position, extra) then
            seen[m] = true
            table.insert(list, { m = m, p = p })
        end
    end

    for _, o in ipairs(workspace:GetDescendants()) do
        if o:IsA("Model") then
            if not Players:GetPlayerFromCharacter(o) then
                if nameMatches(o, keys) then
                    add(hounds, o, 0)
                elseif nameMatches(o, AVOID) then
                    add(avoid, o, avoidExtra)
                end
            end
        elseif o:IsA("TextLabel") and o.Text ~= "" then
            local t = o.Text:lower()
            local list
            for _, key in ipairs(keys) do
                if t:find(key, 1, true) then list = hounds break end
            end
            if not list then
                for _, key in ipairs(AVOID) do
                    if t:find(key, 1, true) then list = avoid break end
                end
            end
            if list then
                local m = o:FindFirstAncestorOfClass("Model")
                if m and m:FindFirstChildOfClass("Humanoid") and not Players:GetPlayerFromCharacter(m) then
                    add(list, m, list == hounds and 0 or avoidExtra)
                end
            end
        end
    end

    cache.hounds, cache.avoid = hounds, avoid

    -- tandai: target = hijau, dihindari = kuning
    if hlFolder then
        for m, h in pairs(hlMap) do
            if not seen[m] or not m.Parent then
                h:Destroy()
                hlMap[m] = nil
            end
        end
        for _, e in ipairs(hounds) do mark(e.m, COLOR_TARGET) end
        for _, e in ipairs(avoid) do mark(e.m, COLOR_AVOID) end
    end
    return cache
end

local function nearest(list, pos)
    local best, bestD
    for _, e in ipairs(list) do
        if e.p and e.p.Parent and alive(e) then
            local d = (e.p.Position - pos).Magnitude
            if not bestD or d < bestD then best, bestD = e, d end
        end
    end
    return best, bestD
end

local function findMob(root)
    local c = scan()
    local best, bestD = nil, num("range")
    for _, e in ipairs(c.hounds) do
        if e.p.Parent and alive(e) and inArea(e.p.Position) then
            local d = (e.p.Position - root.Position).Magnitude
            if d < bestD then
                local safe = true
                for _, a in ipairs(c.avoid) do
                    if a.p.Parent and (a.p.Position - e.p.Position).Magnitude < num("avoid") then
                        safe = false
                        break
                    end
                end
                if safe then best, bestD = e.m, d end
            end
        end
    end
    return best, bestD
end

local function nearestAvoid(root)
    return nearest(scan().avoid, root.Position)
end

local function threatened()
    local _, _, r = getChar()
    if not r then return false end
    local a, d = nearestAvoid(r)
    return a ~= nil and d < num("avoid")
end

-- true hanya jika target sudah dekat (jarak datar <= "Jarak dekat" + 10)
local function houndNear(mob)
    local _, _, r = getChar()
    if not r then return false end
    local limit = num("stop") + 10
    local function within(part)
        local d = Vector3.new(part.Position.X - r.Position.X, 0, part.Position.Z - r.Position.Z).Magnitude
        return d <= limit
    end
    if mob then
        local p = mob.Parent and mobPart(mob)
        return p ~= nil and within(p)
    end
    for _, e in ipairs(scan().hounds) do
        if e.p.Parent and alive(e) and within(e.p) then return true end
    end
    return false
end

-- target hidup terdekat yang sudah dalam jangkauan serang (jarak datar <= "Jarak dekat" + 10)
local function nearHound(root)
    local limit = num("stop") + 10
    local best, bestD
    for _, e in ipairs(scan().hounds) do
        if e.p.Parent and alive(e) then
            local d = Vector3.new(e.p.Position.X - root.Position.X, 0, e.p.Position.Z - root.Position.Z).Magnitude
            if d <= limit and (not bestD or d < bestD) then best, bestD = e, d end
        end
    end
    return best
end

-- satu "tick" serangan (dipakai fight, flee, dodge):
--  * MP cukup & jeda E sudah lewat -> tekan skill E (prioritas)
--  * selama menunggu jeda E        -> klik kiri (tidak diam)
--  * MP minimum (mpRest)           -> klik kiri saja sampai MP pulih
local lastAtkClick = 0
local function attackTick()
    local now = os.clock()
    if not mpRest and now - lastE >= math.max(num("eint"), 0.1) then
        tap(Enum.KeyCode.E)
        lastE = os.clock()
        return
    end
    if num("gapclk") > 0 and now - lastAtkClick >= math.max(num("cint"), 0.05) then
        click()
        lastAtkClick = os.clock()
    end
end

local function sleep(t)
    local s = os.clock()
    while os.clock() - s < t do
        if not running or damaged or recovering or threatened() then return false end
        task.wait(0.03)
    end
    return running and not damaged and not recovering
end

local function rotY(v, deg)
    local r = math.rad(deg)
    local c, s = math.cos(r), math.sin(r)
    return Vector3.new(v.X * c - v.Z * s, 0, v.X * s + v.Z * c)
end

local function rayParams()
    local ignore = {}
    if lp.Character then table.insert(ignore, lp.Character) end
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl.Character then table.insert(ignore, pl.Character) end
    end
    local c = scan()
    for _, e in ipairs(c.hounds) do table.insert(ignore, e.m) end
    for _, e in ipairs(c.avoid) do table.insert(ignore, e.m) end
    local rp = RaycastParams.new()
    rp.FilterType = Enum.RaycastFilterType.Exclude
    rp.FilterDescendantsInstances = ignore
    return rp
end

local function solid(inst)
    return inst:IsA("Terrain") or inst.CanCollide
end

local function blocked(root, dir, rp)
    local dist = num("wall")
    for _, h in ipairs({ 0, -1.5 }) do
        local origin = root.Position + Vector3.new(0, h, 0)
        local res = workspace:Raycast(origin, dir * dist, rp)
        local n = 0
        while res and not solid(res.Instance) and n < 4 do
            local f = rp.FilterDescendantsInstances
            table.insert(f, res.Instance)
            rp.FilterDescendantsInstances = f
            res = workspace:Raycast(origin, dir * dist, rp)
            n += 1
        end
        if res and solid(res.Instance) then return true end
    end
    return false
end

local function navDir(root, desired)
    local now = os.clock()
    local rp = rayParams()

    if now < detour.untilT then
        local d = rotY(desired, detour.ang)
        if not blocked(root, d, rp) then return d, true end
        detour.untilT = 0
    end

    if not blocked(root, desired, rp) then return desired, false end

    local order = { 45, -45, 90, -90, 135, -135, 180 }
    if math.random() < 0.5 then order = { -45, 45, -90, 90, -135, 135, 180 } end
    for _, a in ipairs(order) do
        local d = rotY(desired, a)
        if not blocked(root, d, rp) then
            detour.ang = a
            detour.untilT = now + num("turn")
            return d, true
        end
    end
    detour.ang, detour.untilT = 180, now + num("turn")
    return rotY(desired, 180), true
end

local function checkStuck(root, hum)
    local now = os.clock()
    local flatPos = Vector3.new(root.Position.X, 0, root.Position.Z)
    if not lastPos then
        lastPos, lastPosT = flatPos, now
        return
    end
    if now - lastPosT >= 0.4 then
        if (flatPos - lastPos).Magnitude < 0.7 then
            detour.ang = (math.random() < 0.5) and 90 or -90
            detour.untilT = now + num("turn")
            if hum then hum.Jump = true end
        end
        lastPos, lastPosT = flatPos, now
    end
end

local function steer(dir)
    local cf = workspace.CurrentCamera.CFrame
    local look = Vector3.new(cf.LookVector.X, 0, cf.LookVector.Z)
    local right = Vector3.new(cf.RightVector.X, 0, cf.RightVector.Z)
    look = look.Magnitude > 0 and look.Unit or Vector3.new(0, 0, -1)
    right = right.Magnitude > 0 and right.Unit or Vector3.new(1, 0, 0)
    local f, r = dir:Dot(look), dir:Dot(right)
    local function axis(key, v)
        local on = held[key]
        setKey(key, (on and v > 0.18) or (not on and v > 0.4))
    end
    axis(Enum.KeyCode.W, f)
    axis(Enum.KeyCode.S, -f)
    axis(Enum.KeyCode.D, r)
    axis(Enum.KeyCode.A, -r)
end

local function moveDir(hum, root, dir)
    checkStuck(root, hum)
    local d = navDir(root, dir)
    if mode == 3 then hum:MoveTo(root.Position + d * 12) else steer(d) end
end

local function moveAway(hum, root, fromPos)
    local away = Vector3.new(root.Position.X - fromPos.X, 0, root.Position.Z - fromPos.Z)
    away = away.Magnitude > 0.1 and away.Unit or Vector3.new(1, 0, 0)
    moveDir(hum, root, away)
end

local function approach(mob, sid)
    local t0 = os.clock()
    lockMob = mob
    lastPos = nil
    while running and sid == session do
        if damaged or recovering then releaseAll() return false end
        local _, hum, root = getChar()
        local mp = mob.Parent and mobPart(mob)
        local mh = mob:FindFirstChildOfClass("Humanoid")
        if not root or not mp or (mh and mh.Health <= 0) then releaseAll() return false end

        local a, ad = nearestAvoid(root)
        if a and ad < num("avoid") then releaseAll() return false end

        local diff = mp.Position - root.Position
        local flat = Vector3.new(diff.X, 0, diff.Z)
        if flat.Magnitude <= num("stop") then releaseAll() return true end
        if os.clock() - t0 > 25 then releaseAll() return false end

        checkStuck(root, hum)
        local dir, turning = navDir(root, flat.Unit)
        setStatus(string.format("%s (%.0f stud)", turning and "Belok (halangan)" or "Mendekati", flat.Magnitude))

        if mode == 3 then
            if turning then hum:MoveTo(root.Position + dir * 12) else hum:MoveTo(mp.Position) end
        else
            steer(dir)
        end
        task.wait(0.05)
    end
    releaseAll()
    return false
end

-- Serangan saat dekat target:
--  * MP cukup  -> PRIORITAS skill E tiap "Jeda E" detik. Di sela jeda E -> klik kiri (tidak diam).
--  * MP minimum (mpRest) -> klik kiri saja sampai MP pulih (mencapai "MP lanjut").
-- Tombol "1" TIDAK ditekan di sini (hanya sekali saat START / setelah respawn).
-- Kena damage tidak membatalkan serangan; yang membatalkan: bot stop, mode pulih HP,
-- monster yang dihindari mendekat, target mati, atau target menjauh.
local function fight(mob)
    local function isAlive()
        local h = mob and mob.Parent and mob:FindFirstChildOfClass("Humanoid")
        return mob ~= nil and mob.Parent ~= nil and (not h or h.Health > 0)
    end
    local function canGo()
        return running and not recovering and not threatened() and isAlive()
    end
    local step = math.max(num("cint"), 0.05)

    if not houndNear(mob) then return end -- belum dekat -> jangan mulai

    while canGo() do
        if not houndNear(mob) then return end

        if not mpRest and os.clock() - lastE >= math.max(num("eint"), 0.1) then
            -- MP cukup & jeda E lewat: PRIORITAS skill E
            setStatus("Skill E")
            tap(Enum.KeyCode.E)
            lastE = os.clock()
        else
            -- menunggu jeda E (atau MP minimum): serang pakai klik kiri, jangan diam
            setStatus(mpRest and "MP minimum - klik kiri" or "Tunggu jeda E - klik kiri")
            if num("gapclk") > 0 then click() end
        end

        task.wait(step)
    end
end

-- menghindar setelah kena damage (bukan mode pulih): bergerak sambil menyerang
-- (skill E diprioritaskan, klik hanya saat MP minimum)
local function dodge(sid)
    setStatus("Menghindar + serang!")
    local _, h0 = getChar()
    if h0 then h0.Jump = true end
    local ang = math.rad(math.random(0, 359))
    local base = Vector3.new(math.cos(ang), 0, math.sin(ang))
    lastPos = nil
    local s = os.clock()
    while running and sid == session and os.clock() - s < num("dodge") and not recovering do
        local _, hum, root = getChar()
        if not root then break end
        moveDir(hum, root, base)
        if (num("fleeatk") > 0 or num("dodgeclk") > 0) and houndNear() then
            attackTick()
        end
        task.wait(0.05)
    end
    releaseAll()
    damaged = false
end

local function segDist(p, a, b)
    local ap = Vector3.new(p.X - a.X, 0, p.Z - a.Z)
    local ab = Vector3.new(b.X - a.X, 0, b.Z - a.Z)
    local l2 = ab:Dot(ab)
    if l2 < 1e-3 then return ap.Magnitude end
    local t = math.clamp(ap:Dot(ab) / l2, 0, 1)
    return (ap - ab * t).Magnitude
end

local orbitDir = 1

local function flee(sid)
    lastPos = nil
    local smooth
    local flipLock, stuckPos, stuckT = 0, nil, os.clock()
    local t0, lastScan, mob = os.clock(), 0, nil

    while running and sid == session and not recovering do
        local _, hum, root = getChar()
        if not root then break end
        local a, ad = nearestAvoid(root)
        if not a or ad >= num("avoid") + 30 then break end
        local now = os.clock()
        if now - t0 > 90 then break end

        if now - lastScan > 0.3 then
            lastScan = now
            mob = findMob(root)
        end
        local mp = mob and mob.Parent and mobPart(mob)
        if mp and ad >= num("avoid") then
            local clear = true
            for _, e in ipairs(scan().avoid) do
                if e.p.Parent and segDist(e.p.Position, root.Position, mp.Position) < num("avoid") then
                    clear = false
                    break
                end
            end
            if clear then break end
        end

        local ap = a.p.Position
        local rad = Vector3.new(root.Position.X - ap.X, 0, root.Position.Z - ap.Z)
        local dist = rad.Magnitude
        local out = dist > 0.1 and rad / dist or Vector3.new(1, 0, 0)

        if mp and now > flipLock then
            local tan = Vector3.new(-out.Z, 0, out.X)
            local toMob = Vector3.new(mp.Position.X - root.Position.X, 0, mp.Position.Z - root.Position.Z)
            local want = tan:Dot(toMob) >= 0 and 1 or -1
            if want ~= orbitDir then orbitDir = want flipLock = now + 2 end
        end

        local function compute()
            local tan = Vector3.new(-out.Z, 0, out.X) * orbitDir
            local R = num("avoid") + 8
            local k = math.clamp((R - dist) / 12, -0.6, 1.2)
            local d = tan + out * k
            return d.Magnitude > 0.01 and d.Unit or out
        end

        local desired = compute()
        local rp = rayParams()
        if blocked(root, desired, rp) and now > flipLock then
            orbitDir = -orbitDir
            flipLock = now + 1.5
            smooth = nil
            desired = compute()
            if blocked(root, desired, rp) then desired = navDir(root, desired) end
        end

        smooth = smooth and smooth:Lerp(desired, 0.2) or desired
        if smooth.Magnitude < 0.05 then smooth = desired end
        local dirv = smooth.Unit

        -- kalau ada target dalam jangkauan, kamera diarahkan ke target (bukan ke monster yang dihindari)
        -- supaya klik/skill mengenai target
        local th = num("fleeatk") > 0 and nearHound(root) or nil
        camFocus = th and th.p or a.p
        setStatus(string.format("Mengitari %s (%.0f)%s", a.m.Name, dist,
            th and " + SERANG" or (mp and " - ke target" or " - cari target")))
        if mode == 3 then hum:MoveTo(root.Position + dirv * 12) else steer(dirv) end
        if th then attackTick() end

        local flat = Vector3.new(root.Position.X, 0, root.Position.Z)
        if not stuckPos then
            stuckPos, stuckT = flat, now
        elseif now - stuckT >= 0.6 then
            if (flat - stuckPos).Magnitude < 0.8 then
                orbitDir = -orbitDir
                flipLock = now + 1
                smooth = nil
                hum.Jump = true
            end
            stuckPos, stuckT = flat, now
        end
        task.wait(0.05)
    end
    camFocus = nil
    releaseAll()
end

local function goTo(target, sid, tol, label)
    local t0 = os.clock()
    lastPos = nil
    lockMob = nil
    while running and sid == session do
        local _, hum, root = getChar()
        if not root then releaseAll() return false end

        local diff = target - root.Position
        local flat = Vector3.new(diff.X, 0, diff.Z)
        if flat.Magnitude <= tol then releaseAll() return true end
        if os.clock() - t0 > 60 then releaseAll() return false end

        checkStuck(root, hum)
        local dir, turning = navDir(root, flat.Unit)
        setStatus(string.format("%s (%.0f stud)", label, flat.Magnitude))

        if mode == 3 then
            if turning then hum:MoveTo(root.Position + dir * 12) else hum:MoveTo(target) end
        else
            steer(dir)
        end
        task.wait(0.05)
    end
    releaseAll()
    return false
end

-- MODE PULIH: pergi ke SAFE_POS, diam sampai HP >= batas pulih, lalu kembali ke FARM_POS
local function recover(sid)
    releaseAll()
    lockMob = nil
    damaged = false

    goTo(SAFE_POS, sid, 6, "Ke zona aman")
    damaged = false

    local fleeing = false
    local panicUntil, panicDir = 0, Vector3.new(1, 0, 0)

    local function startFlee()
        if fleeing then return end
        fleeing = true
        releaseAll()
        resetKeyState()
        lastPos = nil
    end

    while running and sid == session do
        local _, hum, root = getChar()
        if not root then break end
        if curHp == nil then break end

        local target = okTarget()
        if curHp >= target then break end

        local c = scan()
        local h, hd = nearest(c.hounds, root.Position)
        local a, ad = nearest(c.avoid, root.Position)
        local danger, dist = h, hd
        if a and (not h or ad < hd) then danger, dist = a, ad end

        local trigger = num("safe") + (fleeing and 20 or 0)
        local now = os.clock()

        if danger and dist < trigger then
            startFlee()
            damaged = false
            setStatus(string.format("MENGHINDAR %d/%d (%.0f)", curHp, target, dist))
            moveAway(hum, root, danger.p.Position)
        elseif damaged or now < panicUntil then
            if now >= panicUntil then
                panicUntil = now + 1.5
                local ang = math.rad(math.random(0, 359))
                panicDir = Vector3.new(math.cos(ang), 0, math.sin(ang))
                damaged = false
            end
            startFlee()
            setStatus(string.format("MENGHINDAR (diserang) %d/%d", curHp, target))
            moveDir(hum, root, panicDir)
        else
            if fleeing then
                releaseAll()
                fleeing = false
            end
            setStatus(string.format("Aman, pulih %d/%d", curHp, target))
        end
        task.wait(0.1)
    end
    releaseAll()

    local _, _, r = getChar()
    if r and running and sid == session then
        goTo(FARM_POS, sid, 6, "Kembali ke titik farm")
    end

    recovering = false
    damaged = false
    lastHp = curHp
    -- (sengaja TIDAK menekan "1" lagi setelah pulih HP; "1" hanya saat START dan setelah respawn)
end

local function cleanText(s)
    return (s:gsub("<[^>]+>", ""):lower())
end

local function guiVisible(o)
    local cur = o
    while cur and cur ~= game do
        if cur:IsA("GuiObject") and not cur.Visible then return false end
        if cur:IsA("ScreenGui") and not cur.Enabled then return false end
        cur = cur.Parent
    end
    return true
end

local function findText(needle, wantButton)
    local pg = lp:FindFirstChild("PlayerGui")
    if not pg then return end
    for _, o in ipairs(pg:GetDescendants()) do
        if not (gui and o:IsDescendantOf(gui)) then
            if (o:IsA("TextLabel") or o:IsA("TextButton")) and o.Text ~= "" then
                if cleanText(o.Text):find(needle, 1, true) and guiVisible(o) then
                    if wantButton then
                        if o:IsA("GuiButton") then return o end
                        return o:FindFirstAncestorWhichIsA("GuiButton") or o
                    end
                    return o
                end
            elseif wantButton and o:IsA("GuiButton") and o.Name:lower():find(needle, 1, true) and guiVisible(o) then
                return o
            end
        end
    end
end

local function findRespawn()
    return findText("respawn", true)
end

local deathCache, deathT = false, 0
local function deathShowing()
    if os.clock() - deathT > 0.5 then
        deathT = os.clock()
        deathCache = findText("you died") ~= nil
    end
    return deathCache
end

local function clickGui(o, useFallback)
    local pos, size = o.AbsolutePosition, o.AbsoluteSize
    local sg = o:FindFirstAncestorWhichIsA("ScreenGui")
    local inset = Vector2.zero
    if not (sg and sg.IgnoreGuiInset) then inset = GuiService:GetGuiInset() end
    local x = pos.X + size.X / 2 + inset.X
    local y = pos.Y + size.Y / 2 + inset.Y
    VIM:SendMouseMoveEvent(x, y, game)
    task.wait(0.05)
    VIM:SendMouseButtonEvent(x, y, 0, true, game, 0)
    task.wait(0.05)
    VIM:SendMouseButtonEvent(x, y, 0, false, game, 0)

    if useFallback and getconnections and o:IsA("GuiButton") then
        pcall(function()
            for _, ev in ipairs({ o.MouseButton1Click, o.Activated }) do
                for _, c in ipairs(getconnections(ev)) do c:Fire() end
            end
        end)
    end
end

local function waitHpFull(sid)
    local t0 = os.clock()
    local moving = false
    task.wait(0.5)
    while running and sid == session do
        local _, hum, root = getChar()
        if root and curHp == nil then break end
        if root then
            local target = curMax or okTarget()
            if curHp >= target - 0.5 then break end
            setStatus(string.format("Menunggu HP penuh %d/%d", curHp, target))

            local c = scan()
            local h, hd = nearest(c.hounds, root.Position)
            local a, ad = nearest(c.avoid, root.Position)
            local danger, dist = h, hd
            if a and (not h or ad < hd) then danger, dist = a, ad end
            if danger and dist < num("safe") then
                moving = true
                moveAway(hum, root, danger.p.Position)
            elseif moving then
                releaseAll()
                moving = false
            end
        else
            setStatus("Menunggu karakter...")
        end
        if os.clock() - t0 > 120 then break end
        task.wait(0.1)
    end
    releaseAll()
end

local function handleDeath(sid)
    releaseAll()
    lockMob = nil
    recovering = false
    damaged = false
    local clicks, lastTry = 0, 0

    while running and sid == session do
        local _, _, root = getChar()
        local died = findText("you died")
        if root and not died then break end

        if died then
            local btn = findRespawn()
            if btn then
                if os.clock() - lastTry >= 1.5 then
                    setStatus("Klik tombol Respawn...")
                    clickGui(btn, clicks >= 1)
                    clicks += 1
                    lastTry = os.clock()
                end
            else
                setStatus("You Died - menunggu tombol Respawn...")
            end
        else
            setStatus("Menunggu karakter...")
        end
        task.wait(0.2)
    end

    if running and sid == session then
        waitHpFull(sid)
    end

    local _, _, r = getChar()
    if r and running and sid == session then
        goTo(FARM_POS, sid, 6, "Ke titik farm")
    end

    -- setelah respawn: tekan Q sekali untuk masuk siaga tempur sebelum berburu
    if running and sid == session and getChar() then
        setStatus("Siaga tempur (Q)...")
        task.wait(0.3)
        tap(Enum.KeyCode.Q)
        task.wait(0.7)
    end

    recovering = false
    damaged = false
    lastHp = curHp
    lockMob = nil
    deathT = 0
    needSkill1 = true -- selesai respawn -> tekan "1" lagi sekali
end

local function mainLoop(sid)
    while running and sid == session do
        local _, hum, root = getChar()
        if not root or deathShowing() then
            handleDeath(sid)
        else
            if lowHp() then recovering = true end

            if recovering then
                recover(sid)
            else
                damaged = false

                -- tekan "1" (aktifkan skill) SEKALI: HANYA saat start / setelah respawn
                if needSkill1 then
                    setStatus("Aktifkan skill 1")
                    tap(Enum.KeyCode.One)
                    needSkill1 = false
                    task.wait(0.3)
                end

                flee(sid)

                setStatus(#targetKeys() == 0 and "Pilih target dulu!" or "Mencari target...")
                local _, _, r2 = getChar()
                local mob, dist = nil, nil
                if r2 then mob, dist = findMob(r2) end
                if not mob then
                    lockMob = nil
                    if #targetKeys() == 0 then
                        setStatus("Pilih target dulu!")
                    else
                        setStatus("Tidak ketemu (range " .. num("range") .. ")")
                    end
                    task.wait(0.7)
                else
                    print("[HuntBot] Target:", mob:GetFullName(), math.floor(dist) .. " stud")
                    if approach(mob, sid) then fight(mob) end
                    if recovering then
                        releaseAll()
                    elseif damaged then
                        dodge(sid)
                    end
                    lockMob = nil
                end
            end
        end
        task.wait()
    end
    lockMob = nil
    releaseAll()
end

local function refreshPoints()
    for i = 1, 2 do
        local p = points[i]
        pointLabels[i].Text = p
            and string.format("T%d: %d, %d, %d", i, p.X, p.Y, p.Z)
            or string.format("T%d: belum diset", i)
        if not coordBoxes[i]:IsFocused() then
            coordBoxes[i].Text = p and string.format("%d, %d, %d", math.floor(p.X + 0.5), math.floor(p.Y + 0.5), math.floor(p.Z + 0.5)) or ""
        end
    end
end

-- terapkan koordinat yang diketik manual (3 angka dipisah koma / spasi)
local function applyCoord(i)
    local v = {}
    for n in coordBoxes[i].Text:gmatch("-?%d+%.?%d*") do
        table.insert(v, tonumber(n))
    end
    if #v ~= 3 then
        setStatus("Koordinat T" .. i .. " tidak valid (format: X, Y, Z)")
        return
    end
    points[i] = Vector3.new(v[1], v[2], v[3])
    refreshPoints()
    drawWalls()
    invalidate() -- scan ulang segera dengan area baru
    setStatus(string.format("Titik %d diset: %d, %d, %d", i, v[1], v[2], v[3]))
end

-- Loop latar belakang: scan ulang tiap "Scan ulang" detik (juga saat bot belum START),
-- bersihkan highlight monster yang sudah mati, dan perbarui info scan.
task.spawn(function()
    while gui and gui.Parent do
        local interval = math.max(num("scanint"), 1)
        pcall(scan)

        -- gambar ulang dinding jika radius / titik / ON-OFF berubah, atau folder hilang (kamera diganti)
        local needWalls = wallsOn and (points[1] or points[2]) and (not wallFolder or not wallFolder.Parent)
        if wallSignature() ~= wallSig or needWalls then
            pcall(drawWalls)
        end

        for m, h in pairs(hlMap) do
            local hum = m.Parent and m:FindFirstChildOfClass("Humanoid")
            if not m.Parent or (hum and hum.Health <= 0) then
                h:Destroy()
                hlMap[m] = nil
            end
        end

        if scanLabel then
            local left = math.max(0, math.ceil(interval - (os.clock() - cache.t)))
            local area = (points[1] or points[2]) and "" or " | area: semua"
            scanLabel.Text = string.format("Scan: %d target(hijau) %d hindari(kuning) | %ds%s",
                #cache.hounds, #cache.avoid, left, area)
        end
        task.wait(1)
    end
end)
